# pour utiliser l'application : ces paquets ne servent qu'à fabriquer la version
# livrée dans les Releases.
#
#   pip install -r requirements.txt -r requirements-build.txt
#   pyinstaller --noconfirm --clean packaging/whiscribe.spec
#
# PyInstaller 6.22 ou plus récent est un minimum réel, pas une précaution : les
# versions antérieures ne savent pas geler numpy 2.5 et l'exécutable produit
# échoue au démarrage sur « numpy._core._exceptions ».

pyinstaller>=6.22,<7
pyinstaller-hooks-contrib>=2026.6

# pip est embarqué dans le gel depuis la version 2.3.0 : c'est lui qui installe
# la séparation des locuteurs dans le dossier d'extensions de l'utilisateur.
# Il est présent dans tout environnement virtuel, la ligne ne sert donc qu'à
# garantir une version qui connaisse « --progress-bar raw », d'où vient la
# progression affichée dans l'interface.
pip>=24.1
